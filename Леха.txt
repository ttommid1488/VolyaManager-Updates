def get_file_version_info(filename)
    try
        if not os.path.isfile(filename) return None
        ver_info_size = ctypes.windll.version.GetFileVersionInfoSizeW(filename, None)
        if not ver_info_size return None
        res_data = ctypes.create_string_buffer(ver_info_size)
        ctypes.windll.version.GetFileVersionInfoW(filename, None, ver_info_size, res_data)
        r = ctypes.c_void_p();
        l = ctypes.c_uint()
        ctypes.windll.version.VerQueryValueW(res_data, , ctypes.byref(r), ctypes.byref(l))

        class VS(ctypes.Structure)
            _fields_ = [('dwSignature', ctypes.c_long), ('dwStrucVersion', ctypes.c_long),
                        ('dwFileVersionMS', ctypes.c_long), ('dwFileVersionLS', ctypes.c_long),
                        ('dwProductVersionMS', ctypes.c_long), ('dwProductVersionLS', ctypes.c_long)]

        ffi = ctypes.cast(r, ctypes.POINTER(VS)).contents
        return f{ffi.dwProductVersionMS  16}.{ffi.dwProductVersionMS & 0xFFFF}
    except
        return None


def get_available_drives()
    drives = []
    bitmask = ctypes.windll.kernel32.GetLogicalDrives()
    for letter in string.ascii_uppercase
        if bitmask & 1 drives.append(f{letter})
        bitmask = 1
    return drives

    def minimize_to_tray(self)
        if pystray
            self.withdraw()
            image = Image.open(assetsicon.ico) if os.path.exists(assetsicon.ico) else Image.new('RGB', (64, 64),
                                                                                                      color='blue')

            menu = (
                item(self.t(tray_show), self.show_from_tray, default=True),
                item(self.t(tray_exit), self.quit_app)
            )
            self.tray_icon = pystray.Icon(VolyaManager, image, Volya Manager, menu)
            # Запускаємо в окремому потоці, щоб не блокувати
            threading.Thread(target=self.tray_icon.run).start()
        else
            # Якщо бібліотеки немає, просто закриваємося як зазвичай (або звертаємо)
            self.iconify()

    def download_remote_catalog(self)
        global ADDON_CATALOG
        url = fhttpsraw.githubusercontent.com{GITHUB_USER}{GITHUB_REPO}maincatalog.json
        try
            req = urllib.request.Request(url, headers={'User-Agent' 'Mozilla5.0'})
            with urllib.request.urlopen(req, timeout=5) as response
                data = json.loads(response.read())
                if data
                    # ЗАХИСТ Якщо на сервері менше аддонів, ніж у нас - не оновлюємо!
                    if len(data)  len(ADDON_CATALOG)
                        ADDON_CATALOG = data
                        self.after(0, self.refresh_ui_list)
                        print(Каталог оновлено з GitHub.)
                    else
                        print(Локальний каталог новіший або такий самий.)
        except Exception as e
            print(fCatalog update failed {e})

    def check_program_update(self)
        try
            api_url = fhttpsapi.github.comrepos{GITHUB_USER}{GITHUB_REPO}releaseslatest
            req = urllib.request.Request(api_url, headers={'User-Agent' 'Mozilla5.0'})
            with urllib.request.urlopen(req) as response
                data = json.loads(response.read())
                remote_tag_raw = data.get(tag_name, )

                clean_remote = remote_tag_raw.lower().replace(v, ).split(-)[0]
                clean_local = APP_VERSION_NUM.lower().replace(v, ).split(-)[0]

                try
                    local_parts = tuple(map(int, clean_local.split(.)))
                    remote_parts = tuple(map(int, clean_remote.split(.)))

                    if remote_parts  local_parts
                        download_url = data[html_url]
                        for asset in data.get(assets, [])
                            if asset[name].endswith(.exe)
                                download_url = asset[browser_download_url]
                                break
                        self.after(0, lambda self.show_update_dialog(remote_tag_raw, download_url))
                except Exception as e
                    print(fError parsing version {e})
        except Exception as e
            print(fAuto-update check failed {e})

    def deep_search_worker(self)
        drives = get_available_drives()
        for drive in drives
            for root, dirs, files in os.walk(drive)
                if $Recycle.Bin in root or Windows in root continue
                candidates = [f for f in files if f.lower().endswith(.exe) and wow in f.lower()]
                if candidates
                    for candidate in candidates
                        full_path = os.path.join(root, candidate)
                        ver = get_file_version_info(full_path)
                        if ver
                            potential_paths = [
                                os.path.join(root, Interface, AddOns),
                                os.path.join(os.path.dirname(root), Interface, AddOns),
                                os.path.join(root, _retail_, Interface, AddOns)
                            ]
                            valid_path = next((p for p in potential_paths if os.path.exists(p)),
                                              os.path.join(root, Interface, AddOns))
                            maj = ver.split(.)[0]
                            exp = next((n for n, p in EXPANSIONS.items() if p == maj), The War Within (11.x))
                            self.game_exe_path = full_path
                            self.after(0, lambda p=valid_path, e=exp, v=ver self.apply_auto_found(p, e, v))
                            return
        self.after(0, self.finish_deep_search_fail)

    def scan_worker(self)
        global ONLINE_CACHE
        prefix = EXPANSIONS[self.selected_expansion.get()]
        total = len(ADDON_CATALOG)
        headers = {'User-Agent' random.choice(USER_AGENTS)}
        auto_update_queue = []

        for i, (name, data) in enumerate(ADDON_CATALOG.items())
            self.progress_bar.set((i + 1)  total)

            is_ignored = name in self.ignored_addons

            if name in ONLINE_CACHE
                res = ONLINE_CACHE[name]
            else
                provider = data.get(provider, github)
                res = {ver None, url None}
                try
                    if provider == github
                        res = self.check_github(data[repo], prefix, headers)
                    elif provider == tukui
                        res = self.check_tukui(data[id], headers)
                    elif provider == wago
                        res = self.check_wago(data[slug], prefix, headers)
                except Exception
                    pass
                ONLINE_CACHE[name] = res

            if self.auto_update_enabled.get() and not is_ignored and res[url]
                loc = self.get_local_ver(data[match])
                if loc and res[ver] and str(res[ver]) not in str(loc)
                    if self.installed_memory.get(name) != res[ver]
                        auto_update_queue.append((name, res[url], res[ver]))

        self.save_cache()
        self.after(0, lambda self.finish_scan(auto_update_queue))

    @staticmethod
    def check_github(repo, prefix, headers)
        api = fhttpsapi.github.comrepos{repo}releases
        req = urllib.request.Request(api, headers=headers)
        with urllib.request.urlopen(req) as r
            rels = json.loads(r.read())
            for rel in rels
                tag = rel[tag_name]
                clean = tag.lower().replace(v, ).replace(release, ).strip()
                if clean.startswith(prefix)
                    url = rel[zipball_url]
                    for a in rel.get(assets, [])
                        if a[name].endswith(.zip)
                            url = a[browser_download_url]
                            break
                    return {ver tag, url url}
            if prefix == 11 and rels
                latest = rels[0]
                url = latest[zipball_url]
                for a in latest.get(assets, [])
                    if a[name].endswith(.zip)
                        url = a[browser_download_url]
                        break
                return {ver latest[tag_name], url url}
        return {ver None, url None}

    @staticmethod
    def check_tukui(ui_id, headers)
        url = httpswww.tukui.orgapi.phpui=elvui if ui_id == -1 else httpswww.tukui.orgapi.phpui=tukui
        req = urllib.request.Request(url, headers=headers)
        with urllib.request.urlopen(req) as r
            data = json.loads(r.read())
            if data.get(version)
                return {ver data.get(version), url data.get(url)}
        return {ver None, url None}

    @staticmethod
    def check_wago(slug, prefix, headers)
        url_ver = fhttpsaddons.wago.ioapiprojects{slug}versionstability=stable
        req_ver = urllib.request.Request(url_ver, headers=headers)
        with urllib.request.urlopen(req_ver) as r
            versions = json.loads(r.read())
            for v in versions
                supported_vers = v.get(supported_retail_patches, []) or v.get(gameVersion, [])
                for sv in supported_vers
                    if sv.startswith(prefix)
                        return {ver v.get(version), url v.get(url)}
            if prefix == 11 and len(versions)  0
                return {ver versions[0].get(version), url versions[0].get(url)}
        return {ver None, url None}

    def install(self, url, name, remote_ver, is_batch=False)
        self.status.configure(text=f{self.t('status_installing')} {name}..., text_color=COLOR_YELLOW)
        if self.btn_scan
            self.btn_scan.configure(state=disabled)
        threading.Thread(target=self.install_logic, args=(url, name, remote_ver, is_batch)).start()

    def install_logic(self, url, name, remote_ver, is_batch=False)
        try
            req = urllib.request.Request(url, headers={'User-Agent' 'Mozilla5.0'})
            with urllib.request.urlopen(req) as response
                total_size = int(response.info().get('Content-Length', 0))
                downloaded = 0
                chunk_size = 8192
                data_io = io.BytesIO()
                while True
                    chunk = response.read(chunk_size)
                    if not chunk break
                    data_io.write(chunk)
                    downloaded += len(chunk)
                    if total_size  0 and not is_batch
                        self.after(0, lambda p=downloaded  total_size self.progress_bar.set(p))

            temp = os.path.join(self.addon_path.get(), _temp)
            if os.path.exists(temp) shutil.rmtree(temp)
            os.makedirs(temp, exist_ok=True)
            with zipfile.ZipFile(data_io) as z
                z.extractall(temp)
            src = temp
            items = os.listdir(temp)
            if len(items) == 1 and os.path.isdir(os.path.join(temp, items[0]))
                if name not in [ElvUI, TukUI, Cell, VuhDo, Deadly Boss Mods] src = os.path.join(temp,
                                                                                                           items[0])
            dst = self.addon_path.get()
            for item in os.listdir(src)
                if item.startswith(.) or readme in item.lower() continue
                s = os.path.join(src, item);
                d = os.path.join(dst, item)
                if os.path.exists(d)
                    if os.path.isdir(d)
                        shutil.rmtree(d)
                    else
                        os.remove(d)
                if os.path.isdir(s)
                    shutil.copytree(s, d)
                else
                    shutil.copy2(s, d)
            shutil.rmtree(temp)

            self.installed_memory[name] = remote_ver
            self.save_settings()
            RECENTLY_UPDATED.add(name)

            if not is_batch
                self.after(0, lambda self.set_temporary_status(f✅ {name} {self.t('msg_install_ok')}, COLOR_GREEN))
                self.after(0, self.refresh_ui_list)

        except Exception as e
            if not is_batch
                if Permission in str(e) or access in str(e)
                    self.after(0, lambda messagebox.showwarning(Error, self.t(msg_close_wow)))
                else
                    self.after(0, lambda messagebox.showerror(Error, str(e)))
        finally
            if not is_batch
                if self.btn_scan
                    self.after(0, lambda self.btn_scan.configure(state=normal))
                self.after(0, lambda self.progress_bar.set(0))

    def delete_addon(self, name)
        if not messagebox.askyesno(self.t(btn_delete_title), f{self.t('btn_delete_confirm')} {name}) return
        match_folder = ADDON_CATALOG[name][match]
        main_dir = os.path.join(self.addon_path.get(), match_folder)
        extras = ADDON_CATALOG[name].get(extra_folders, [])
        try
            if os.path.exists(main_dir) shutil.rmtree(main_dir)
            for extra in extras
                extra_path = os.path.join(self.addon_path.get(), extra)
                if os.path.exists(extra_path) shutil.rmtree(extra_path)

            if name in self.installed_memory
                del self.installed_memory[name]
                self.save_settings()
            if name in RECENTLY_UPDATED RECENTLY_UPDATED.remove(name)

            self.refresh_ui_list()
        except Exception as e
            messagebox.showerror(Error, str(e))

    def minimize_to_tray(self)
        if pystray
            self.withdraw()
            image = Image.open(assetsicon.ico) if os.path.exists(assetsicon.ico) else Image.new('RGB', (64, 64),
                                                                                                      color='blue')

            menu = (
                item(self.t(tray_show), self.show_from_tray, default=True),
                item(self.t(tray_exit), self.quit_app)
            )
            self.tray_icon = pystray.Icon(VolyaManager, image, Volya Manager, menu)
            # Запускаємо в окремому потоці
            threading.Thread(target=self.tray_icon.run).start()
        else
            self.iconify()

    def show_from_tray(self)
        if self.tray_icon
            self.tray_icon.stop()
        self.after(0, self.deiconify)

    def quit_app(self)
        if self.tray_icon
            self.tray_icon.stop()
        self.quit()
